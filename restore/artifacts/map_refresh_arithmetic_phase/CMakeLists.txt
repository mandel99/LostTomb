cmake_minimum_required(VERSION 3.20)
project(map_refresh_arithmetic C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(map_refresh_numeric numeric_fixture.c map_refresh_fp.c)
target_compile_definitions(map_refresh_numeric PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(map_refresh_numeric PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
foreach(kind IN ITEMS trace plain)
 add_executable(map_refresh_weighted_${kind} weighted_terrain_fixture.c unit_01192.c unit_01193.c map_refresh_fp.c
  "${RESTORE}/artifacts/map_refresh_phase/unit_01198.c"
  "${RESTORE}/artifacts/map_refresh_phase/x87_conversion/map_x87_conversion.c")
 target_include_directories(map_refresh_weighted_${kind} PRIVATE "${RESTORE}/include" "${RESTORE}/artifacts/map_refresh_phase" "${RESTORE}/artifacts/map_refresh_phase/x87_conversion")
 target_compile_definitions(map_refresh_weighted_${kind} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(map_refresh_weighted_${kind} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 set_property(TARGET map_refresh_weighted_${kind} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
 target_link_libraries(map_refresh_weighted_${kind} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 target_link_options(map_refresh_weighted_${kind} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()
target_compile_definitions(map_refresh_weighted_trace PRIVATE MAP_REFRESH_TRACE MAP_REFRESH_ARITHMETIC_TRACE)
enable_testing()
add_test(NAME map_refresh_numeric_launch COMMAND map_refresh_numeric)
add_test(NAME map_refresh_weighted_trace_launch COMMAND map_refresh_weighted_trace)
add_test(NAME map_refresh_weighted_plain_launch COMMAND map_refresh_weighted_plain)
