cmake_minimum_required(VERSION 3.20)
project(editor_validation_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PHASE "${RESTORE}/artifacts/editor_validation_phase")
add_executable(editor_validation_verify "${PHASE}/tests/editor_validation.c" "${PHASE}/tests/editor_actual_trace.c" "${PHASE}/tests/editor_actual_plain.c")
target_include_directories(editor_validation_verify PRIVATE "${PHASE}/include" "${PHASE}/tests" "${RESTORE}/include")
target_compile_definitions(editor_validation_verify PRIVATE _CRT_SECURE_NO_WARNINGS EDITOR_VALIDATION_STAGED)
target_compile_options(editor_validation_verify PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
set_property(TARGET editor_validation_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(editor_validation_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(editor_validation_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME editor_validation_smoke COMMAND editor_validation_verify)
