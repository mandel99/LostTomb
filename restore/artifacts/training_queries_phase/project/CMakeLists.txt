cmake_minimum_required(VERSION 3.20)
project(training_queries_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PHASE "${RESTORE}/artifacts/training_queries_phase")
add_executable(training_queries_verify "${PHASE}/tests/training_queries.c" "${PHASE}/tests/queries_actual.c")
target_include_directories(training_queries_verify PRIVATE "${PHASE}/include" "${PHASE}/tests" "${RESTORE}/include")
target_compile_definitions(training_queries_verify PRIVATE _CRT_SECURE_NO_WARNINGS TRAINING_QUERIES_STAGED)
target_compile_options(training_queries_verify PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
set_property(TARGET training_queries_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(training_queries_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(training_queries_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME training_queries_smoke COMMAND training_queries_verify)
