cmake_minimum_required(VERSION 3.20)
project(map_neighbors_phase C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
  add_executable(map_neighbors_${kind} map_neighbors_fixture.c unit_01138.c unit_01139.c unit_01140.c unit_01141.c unit_01142.c "${RESTORE}/src/unit_01137.c")
  target_include_directories(map_neighbors_${kind} PRIVATE "${RESTORE}/include")
  target_compile_options(map_neighbors_${kind} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
  target_compile_definitions(map_neighbors_${kind} PRIVATE _CRT_SECURE_NO_WARNINGS)
  set_property(TARGET map_neighbors_${kind} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(map_neighbors_${kind} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
  target_link_options(map_neighbors_${kind} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
endforeach()
target_compile_definitions(map_neighbors_trace PRIVATE MAP_NEIGHBORS_TRACE)
enable_testing()
add_test(NAME map_neighbors_trace_launch COMMAND map_neighbors_trace)
add_test(NAME map_neighbors_plain_launch COMMAND map_neighbors_plain)
