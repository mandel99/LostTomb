cmake_minimum_required(VERSION 3.20)
if(NOT TARGET pharaoh_restored)
 project(map_menu_scroll C)
 set(MSS_PRIVATE TRUE)
endif()
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 if(MSS_PRIVATE)
  set(MSS_TARGET map_scroll_${kind})
  set(MSS_SRC "${CMAKE_CURRENT_LIST_DIR}")
 else()
  set(MSS_TARGET restore_map_menu_scroll_${kind}_verify)
  set(MSS_SRC "${RESTORE}/src")
 endif()
 add_executable(${MSS_TARGET} "${CMAKE_CURRENT_LIST_DIR}/record_fixture.c" "${MSS_SRC}/unit_00609.c" "${MSS_SRC}/unit_00610.c")
 target_include_directories(${MSS_TARGET} PRIVATE "${MSS_SRC}/../include" "${CMAKE_CURRENT_LIST_DIR}" "${RESTORE}/include")
 target_compile_definitions(${MSS_TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(${MSS_TARGET} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 if(kind STREQUAL "trace")
  target_compile_definitions(${MSS_TARGET} PRIVATE MAP_MENU_SCROLL_TRACE)
 endif()
 if(MSS_PRIVATE)
  set_property(TARGET ${MSS_TARGET} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(${MSS_TARGET} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 else()
  target_link_libraries(${MSS_TARGET} PRIVATE restore_data advapi32)
  add_test(NAME ${MSS_TARGET} COMMAND ${MSS_TARGET})
 endif()
 target_link_options(${MSS_TARGET} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()
