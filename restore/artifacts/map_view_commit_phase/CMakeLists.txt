cmake_minimum_required(VERSION 3.20)
if(NOT TARGET pharaoh_restored)
 project(map_view_commit C)
 set(MC_PRIVATE TRUE)
endif()
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 if(MC_PRIVATE)
  set(MC_TARGET map_commit_${kind})
  set(MC_SRC "${CMAKE_CURRENT_LIST_DIR}")
 else()
  set(MC_TARGET restore_map_view_commit_${kind}_verify)
  set(MC_SRC "${RESTORE}/src")
 endif()
 add_executable(${MC_TARGET} "${CMAKE_CURRENT_LIST_DIR}/record_fixture.c" "${MC_SRC}/unit_00070.c" "${MC_SRC}/unit_00076.c" "${RESTORE}/src/unit_00610.c")
 target_include_directories(${MC_TARGET} PRIVATE "${MC_SRC}/../include" "${CMAKE_CURRENT_LIST_DIR}" "${RESTORE}/include")
 target_compile_definitions(${MC_TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(${MC_TARGET} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 if(kind STREQUAL "trace")
  target_compile_definitions(${MC_TARGET} PRIVATE MAP_VIEW_COMMIT_TRACE MAP_MENU_SCROLL_TRACE)
 endif()
 if(MC_PRIVATE)
  set_property(TARGET ${MC_TARGET} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(${MC_TARGET} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 else()
  target_link_libraries(${MC_TARGET} PRIVATE restore_data advapi32)
  add_test(NAME ${MC_TARGET} COMMAND ${MC_TARGET})
 endif()
 target_link_options(${MC_TARGET} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()
