cmake_minimum_required(VERSION 3.20)
if(NOT TARGET pharaoh_restored)
 project(map_record_refresh C)
 set(MRR_PRIVATE TRUE)
endif()
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 if(MRR_PRIVATE)
  set(MRR_TARGET map_record_${kind})
  set(MRR_SRC "${CMAKE_CURRENT_LIST_DIR}")
 else()
  set(MRR_TARGET restore_map_record_refresh_${kind}_verify)
  set(MRR_SRC "${RESTORE}/src")
 endif()
 add_executable(${MRR_TARGET} "${CMAKE_CURRENT_LIST_DIR}/record_fixture.c" "${MRR_SRC}/unit_00606.c" "${MRR_SRC}/unit_00608.c")
 target_include_directories(${MRR_TARGET} PRIVATE "${MRR_SRC}/../include" "${CMAKE_CURRENT_LIST_DIR}" "${RESTORE}/include")
 target_compile_definitions(${MRR_TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(${MRR_TARGET} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 if(kind STREQUAL "trace")
  target_compile_definitions(${MRR_TARGET} PRIVATE MAP_RECORD_REFRESH_TRACE)
 endif()
 if(MRR_PRIVATE)
  set_property(TARGET ${MRR_TARGET} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(${MRR_TARGET} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 else()
  target_link_libraries(${MRR_TARGET} PRIVATE restore_data advapi32)
  add_test(NAME ${MRR_TARGET} COMMAND ${MRR_TARGET})
 endif()
 target_link_options(${MRR_TARGET} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()
