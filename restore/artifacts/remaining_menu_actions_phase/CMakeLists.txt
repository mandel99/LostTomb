cmake_minimum_required(VERSION 3.20)
project(remaining_menu_actions C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 add_executable(remaining_menu_${kind} unit_00049.c unit_00077.c remaining_menu_route.c remaining_menu_fixture.c "${RESTORE}/src/unit_01628.c")
 target_include_directories(remaining_menu_${kind} PRIVATE "${RESTORE}/include")
 target_compile_definitions(remaining_menu_${kind} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(remaining_menu_${kind} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 set_property(TARGET remaining_menu_${kind} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
 target_link_libraries(remaining_menu_${kind} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 target_link_options(remaining_menu_${kind} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
endforeach()
target_compile_definitions(remaining_menu_trace PRIVATE REMAINING_MENU_TRACE)
enable_testing()
add_test(NAME remaining_menu_trace_launch COMMAND remaining_menu_trace)
add_test(NAME remaining_menu_plain_launch COMMAND remaining_menu_plain)
add_executable(remaining_menu_production_link_probe production_link_probe.c unit_00049.c unit_00077.c remaining_menu_route.c)
target_include_directories(remaining_menu_production_link_probe PRIVATE "${RESTORE}/include")
target_compile_options(remaining_menu_production_link_probe PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
set_property(TARGET remaining_menu_production_link_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(remaining_menu_production_link_probe PRIVATE
 "${RESTORE}/build/c-only-debug/Debug/pharaoh_restored.lib"
 "${RESTORE}/build/c-only-debug/Debug/restore_data.lib"
 kernel32 user32 gdi32 shell32 ole32 oleaut32 advapi32 winmm ddraw dxguid legacy_stdio_definitions)
target_link_options(remaining_menu_production_link_probe PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
