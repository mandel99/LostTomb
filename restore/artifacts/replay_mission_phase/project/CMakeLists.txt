cmake_minimum_required(VERSION 3.20)
project(replay_mission_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PHASE "${RESTORE}/artifacts/replay_mission_phase")
add_executable(replay_mission_verify "${PHASE}/tests/replay_mission.c" "${PHASE}/tests/replay_actual_trace.c" "${PHASE}/tests/replay_actual_plain.c")
target_include_directories(replay_mission_verify PRIVATE "${PHASE}/include" "${PHASE}/tests" "${RESTORE}/include")
target_compile_definitions(replay_mission_verify PRIVATE _CRT_SECURE_NO_WARNINGS REPLAY_MISSION_STAGED)
target_compile_options(replay_mission_verify PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
set_property(TARGET replay_mission_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(replay_mission_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(replay_mission_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME replay_mission_smoke COMMAND replay_mission_verify)
