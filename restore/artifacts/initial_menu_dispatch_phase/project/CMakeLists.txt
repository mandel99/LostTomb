cmake_minimum_required(VERSION 3.25)
project(InitialMenuDispatch LANGUAGES C)
get_filename_component(ROOT "../../.." ABSOLUTE)
get_filename_component(PHASE ".." ABSOLUTE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(restore_initial_menu_dispatch_verify "${PHASE}/tests/initial_dispatch.c" "${PHASE}/tests/actual_trace.c" "${PHASE}/tests/actual_plain.c")
target_include_directories(restore_initial_menu_dispatch_verify PRIVATE "${PHASE}/include" "${ROOT}/include" "${PHASE}/tests")
target_compile_definitions(restore_initial_menu_dispatch_verify PRIVATE INITIAL_DISPATCH_STAGED _CRT_SECURE_NO_WARNINGS)
target_compile_options(restore_initial_menu_dispatch_verify PRIVATE /W4 /WX $<$<CONFIG:Debug>:/Od> $<$<CONFIG:Release>:/O2>)
target_link_libraries(restore_initial_menu_dispatch_verify PRIVATE "$<IF:$<CONFIG:Debug>,${ROOT}/build/c-only-debug/Debug/restore_data.lib,${ROOT}/build/c-only/Release/restore_data.lib>")
target_link_options(restore_initial_menu_dispatch_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_library(initial_dispatch_training_compile STATIC "${PHASE}/runtime/initial_menu_dispatch_boundary.c")
target_include_directories(initial_dispatch_training_compile PRIVATE "${PHASE}/include" "${ROOT}/include")
target_compile_options(initial_dispatch_training_compile PRIVATE /W4 /WX $<$<CONFIG:Debug>:/Od> $<$<CONFIG:Release>:/O2>)
add_test(NAME initial_dispatch_smoke COMMAND restore_initial_menu_dispatch_verify)
