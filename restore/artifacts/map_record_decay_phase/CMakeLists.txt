cmake_minimum_required(VERSION 3.20)
if(NOT TARGET pharaoh_restored)
 project(map_record_decay C)
 set(MDR_PRIVATE TRUE)
endif()
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 if(MDR_PRIVATE)
  set(MDR_TARGET map_decay_${kind})
  set(MDR_SRC "${CMAKE_CURRENT_LIST_DIR}")
 else()
  set(MDR_TARGET restore_map_record_decay_${kind}_verify)
  set(MDR_SRC "${RESTORE}/src")
 endif()
 add_executable(${MDR_TARGET} "${CMAKE_CURRENT_LIST_DIR}/record_fixture.c" "${MDR_SRC}/unit_00613.c" "${MDR_SRC}/unit_00614.c")
 target_include_directories(${MDR_TARGET} PRIVATE "${MDR_SRC}/../include" "${CMAKE_CURRENT_LIST_DIR}" "${RESTORE}/include")
 target_compile_definitions(${MDR_TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(${MDR_TARGET} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 if(kind STREQUAL "trace")
  target_compile_definitions(${MDR_TARGET} PRIVATE MAP_RECORD_DECAY_TRACE)
 endif()
 if(MDR_PRIVATE)
  set_property(TARGET ${MDR_TARGET} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(${MDR_TARGET} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 else()
  target_link_libraries(${MDR_TARGET} PRIVATE restore_data advapi32)
  add_test(NAME ${MDR_TARGET} COMMAND ${MDR_TARGET})
 endif()
 target_link_options(${MDR_TARGET} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()
