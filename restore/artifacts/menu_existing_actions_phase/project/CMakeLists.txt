cmake_minimum_required(VERSION 3.20)
project(menu_existing_actions_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PHASE "${RESTORE}/artifacts/menu_existing_actions_phase")
add_executable(menu_existing_actions_verify "${PHASE}/tests/menu_existing_actions.c" "${PHASE}/tests/existing_actual_entries.c" "${PHASE}/tests/existing_actual_loadsave.c" "${PHASE}/tests/existing_actual_helpers.c" "${PHASE}/tests/unreached_link_sentinels.c")
target_include_directories(menu_existing_actions_verify PRIVATE "${PHASE}/tests" "${RESTORE}/include")
target_compile_definitions(menu_existing_actions_verify PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(menu_existing_actions_verify PRIVATE /W4 /Gy "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
set_property(TARGET menu_existing_actions_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(menu_existing_actions_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(menu_existing_actions_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /OPT:REF /INCREMENTAL:NO)
enable_testing()
add_test(NAME menu_existing_actions_smoke COMMAND menu_existing_actions_verify)
