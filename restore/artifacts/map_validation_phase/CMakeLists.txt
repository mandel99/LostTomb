cmake_minimum_required(VERSION 3.20)
if(NOT TARGET pharaoh_restored)
 project(map_validation C)
 set(MV_PRIVATE TRUE)
endif()
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 if(MV_PRIVATE)
  set(MV_TARGET map_validation_${kind})
  set(MV_SRC "${CMAKE_CURRENT_LIST_DIR}")
 else()
  set(MV_TARGET restore_map_validation_${kind}_verify)
  set(MV_SRC "${RESTORE}/src")
 endif()
 add_executable(${MV_TARGET} "${CMAKE_CURRENT_LIST_DIR}/record_fixture.c" "${MV_SRC}/unit_00093.c" "${MV_SRC}/unit_00611.c" "${MV_SRC}/unit_00612.c" "${MV_SRC}/unit_00615.c" "${MV_SRC}/unit_00622.c" "${MV_SRC}/unit_00623.c" "${MV_SRC}/unit_00628.c")
 target_include_directories(${MV_TARGET} PRIVATE "${MV_SRC}/../include" "${CMAKE_CURRENT_LIST_DIR}" "${RESTORE}/include")
 target_compile_definitions(${MV_TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS FUN_005344c0=map_validation_message_boundary)
 target_compile_options(${MV_TARGET} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 if(kind STREQUAL "trace")
  target_compile_definitions(${MV_TARGET} PRIVATE MAP_VALIDATION_TRACE)
 endif()
 if(MV_PRIVATE)
  set_property(TARGET ${MV_TARGET} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(${MV_TARGET} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 else()
  target_link_libraries(${MV_TARGET} PRIVATE restore_data advapi32)
  add_test(NAME ${MV_TARGET} COMMAND ${MV_TARGET})
 endif()
 target_link_options(${MV_TARGET} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()

if(MV_PRIVATE)
 add_executable(map_validation_link_probe link_probe.c unit_00093.c unit_00611.c unit_00612.c unit_00615.c unit_00622.c unit_00623.c unit_00628.c)
 target_include_directories(map_validation_link_probe PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${RESTORE}/include")
 set_property(TARGET map_validation_link_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
 target_link_libraries(map_validation_link_probe PRIVATE "${RESTORE}/build/c-only-debug/Debug/pharaoh_restored.lib" "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" kernel32 user32 gdi32 shell32 ole32 oleaut32 advapi32 winmm ddraw dxguid legacy_stdio_definitions)
 target_link_options(map_validation_link_probe PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endif()
