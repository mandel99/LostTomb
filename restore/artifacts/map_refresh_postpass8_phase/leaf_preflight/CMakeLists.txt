cmake_minimum_required(VERSION 3.20)
project(map8_leaf C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(map8_leaf fixture.c "${RESTORE}/src/unit_01235.c" "${RESTORE}/src/unit_01238.c" "${RESTORE}/src/unit_01098.c")
target_include_directories(map8_leaf PRIVATE "${RESTORE}/include")
target_compile_definitions(map8_leaf PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(map8_leaf PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
set_property(TARGET map8_leaf PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(map8_leaf PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(map8_leaf PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
