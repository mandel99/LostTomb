cmake_minimum_required(VERSION 3.20)
project(settings_callbacks_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PHASE "${RESTORE}/artifacts/settings_callbacks_phase")
add_executable(settings_callbacks_verify "${PHASE}/tests/settings_callbacks.c" "${PHASE}/tests/settings_actual_trace.c" "${PHASE}/tests/settings_actual_plain.c")
target_include_directories(settings_callbacks_verify PRIVATE "${PHASE}/include" "${PHASE}/tests" "${RESTORE}/include")
target_compile_definitions(settings_callbacks_verify PRIVATE _CRT_SECURE_NO_WARNINGS SETTINGS_CALLBACKS_STAGED)
target_compile_options(settings_callbacks_verify PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
set_property(TARGET settings_callbacks_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(settings_callbacks_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(settings_callbacks_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME settings_callbacks_smoke COMMAND settings_callbacks_verify)
