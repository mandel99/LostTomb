cmake_minimum_required(VERSION 3.20)
project(training_panel_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PHASE "${RESTORE}/artifacts/training_panel_phase")
add_executable(training_panel_verify "${PHASE}/tests/training_panel.c" "${PHASE}/tests/training_actual_trace.c" "${PHASE}/tests/training_actual_plain.c")
target_include_directories(training_panel_verify PRIVATE "${PHASE}/include" "${PHASE}/tests" "${RESTORE}/include")
target_compile_definitions(training_panel_verify PRIVATE _CRT_SECURE_NO_WARNINGS TRAINING_PANEL_STAGED)
target_compile_options(training_panel_verify PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
set_property(TARGET training_panel_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(training_panel_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(training_panel_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME training_panel_smoke COMMAND training_panel_verify)
