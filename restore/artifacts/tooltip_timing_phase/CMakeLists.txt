cmake_minimum_required(VERSION 3.20)
if(NOT TARGET pharaoh_restored)
 project(tooltip_timing C)
 set(TT_PRIVATE TRUE)
endif()
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 if(TT_PRIVATE)
  set(TT_TARGET tooltip_${kind})
  set(TT_SRC "${CMAKE_CURRENT_LIST_DIR}")
 else()
  set(TT_TARGET restore_tooltip_timing_${kind}_verify)
  set(TT_SRC "${RESTORE}/src")
 endif()
 add_executable(${TT_TARGET} "${CMAKE_CURRENT_LIST_DIR}/record_fixture.c" "${TT_SRC}/unit_00040.c" "${TT_SRC}/unit_00073.c")
 target_include_directories(${TT_TARGET} PRIVATE "${TT_SRC}/../include" "${CMAKE_CURRENT_LIST_DIR}" "${RESTORE}/include")
 target_compile_definitions(${TT_TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(${TT_TARGET} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 if(kind STREQUAL "trace")
  target_compile_definitions(${TT_TARGET} PRIVATE TOOLTIP_TIMING_TRACE)
 endif()
 if(TT_PRIVATE)
  set_property(TARGET ${TT_TARGET} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(${TT_TARGET} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 else()
  target_link_libraries(${TT_TARGET} PRIVATE restore_data advapi32)
  add_test(NAME ${TT_TARGET} COMMAND ${TT_TARGET})
 endif()
 target_link_options(${TT_TARGET} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()
