cmake_minimum_required(VERSION 3.20)
if(NOT TARGET pharaoh_restored)
 project(map_hover_geometry C)
 set(MH_PRIVATE TRUE)
endif()
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 if(MH_PRIVATE)
  set(MH_TARGET map_hover_${kind})
  set(MH_SRC "${CMAKE_CURRENT_LIST_DIR}")
 else()
  set(MH_TARGET restore_map_hover_geometry_${kind}_verify)
  set(MH_SRC "${RESTORE}/src")
 endif()
 add_executable(${MH_TARGET} "${CMAKE_CURRENT_LIST_DIR}/record_fixture.c" "${MH_SRC}/unit_00460.c")
 target_include_directories(${MH_TARGET} PRIVATE "${MH_SRC}/../include" "${CMAKE_CURRENT_LIST_DIR}" "${RESTORE}/include")
 target_compile_definitions(${MH_TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(${MH_TARGET} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 if(kind STREQUAL "trace")
  target_compile_definitions(${MH_TARGET} PRIVATE MAP_HOVER_GEOMETRY_TRACE)
 endif()
 if(MH_PRIVATE)
  set_property(TARGET ${MH_TARGET} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(${MH_TARGET} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 else()
  target_link_libraries(${MH_TARGET} PRIVATE restore_data advapi32)
  add_test(NAME ${MH_TARGET} COMMAND ${MH_TARGET})
 endif()
 target_link_options(${MH_TARGET} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()
