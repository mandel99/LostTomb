cmake_minimum_required(VERSION 3.20)
if(NOT TARGET pharaoh_restored)
 project(map_file_load C)
 set(MF_PRIVATE TRUE)
endif()
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 if(MF_PRIVATE)
  set(MF_TARGET map_file_${kind})
  set(MF_SRC "${CMAKE_CURRENT_LIST_DIR}")
 else()
  set(MF_TARGET restore_map_file_load_${kind}_verify)
  set(MF_SRC "${RESTORE}/src")
 endif()
 add_executable(${MF_TARGET} "${CMAKE_CURRENT_LIST_DIR}/record_fixture.c" "${MF_SRC}/unit_00072.c" "${MF_SRC}/unit_01740.c" "${MF_SRC}/unit_01741.c" "${MF_SRC}/unit_01747.c" "${RESTORE}/src/unit_01663.c" "${RESTORE}/src/unit_01664.c" "${RESTORE}/src/unit_01743.c" "${RESTORE}/src/unit_01745.c" "${RESTORE}/src/unit_01746.c")
 target_include_directories(${MF_TARGET} PRIVATE "${MF_SRC}/../include" "${CMAKE_CURRENT_LIST_DIR}" "${RESTORE}/include")
 target_compile_definitions(${MF_TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS MAP_FILE_LOAD_PROOF FUN_004d5520=map_file_deep_load FUN_004de600=map_file_deep_start FUN_004ce6e0=map_file_diagnostic)
 target_compile_options(${MF_TARGET} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 if(kind STREQUAL "trace")
  target_compile_definitions(${MF_TARGET} PRIVATE MAP_FILE_LOAD_TRACE)
 endif()
 if(MF_PRIVATE)
  set_property(TARGET ${MF_TARGET} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(${MF_TARGET} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 else()
  target_link_libraries(${MF_TARGET} PRIVATE restore_data advapi32)
  add_test(NAME ${MF_TARGET} COMMAND ${MF_TARGET})
 endif()
 target_link_options(${MF_TARGET} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()

if(MF_PRIVATE)
 add_executable(map_file_link_probe link_probe.c unit_00072.c unit_01740.c unit_01741.c unit_01747.c)
 target_include_directories(map_file_link_probe PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${RESTORE}/include")
 set_property(TARGET map_file_link_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
 target_compile_definitions(map_file_link_probe PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_link_libraries(map_file_link_probe PRIVATE "${RESTORE}/build/c-only-debug/Debug/pharaoh_restored.lib" "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" kernel32 user32 gdi32 shell32 ole32 oleaut32 advapi32 winmm ddraw dxguid legacy_stdio_definitions)
 target_link_options(map_file_link_probe PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endif()
