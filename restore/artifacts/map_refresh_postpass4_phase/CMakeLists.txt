cmake_minimum_required(VERSION 3.20)
project(map_refresh_postpass4 C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 add_executable(map_post4_${kind} postpass4_fixture.c unit_01149.c unit_01152.c unit_01153.c)
 target_include_directories(map_post4_${kind} PRIVATE "${RESTORE}/include")
 target_compile_definitions(map_post4_${kind} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(map_post4_${kind} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 set_property(TARGET map_post4_${kind} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
 target_link_libraries(map_post4_${kind} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 target_link_options(map_post4_${kind} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()
target_compile_definitions(map_post4_trace PRIVATE MAP_REFRESH_POSTPASS4_TRACE)
enable_testing()
add_test(NAME map_post4_trace_launch COMMAND map_post4_trace)
add_test(NAME map_post4_plain_launch COMMAND map_post4_plain)
