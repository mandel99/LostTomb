cmake_minimum_required(VERSION 3.20)
project(map_refresh_phase C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
 add_executable(map_refresh_leaf_${kind} map_refresh_leaf_fixture.c unit_01198.c)
 target_include_directories(map_refresh_leaf_${kind} PRIVATE "${RESTORE}/include")
 target_compile_definitions(map_refresh_leaf_${kind} PRIVATE _CRT_SECURE_NO_WARNINGS)
 target_compile_options(map_refresh_leaf_${kind} PRIVATE /W4 "$<$<CONFIG:Debug>:/Od>" "$<$<CONFIG:Release>:/O2>")
 set_property(TARGET map_refresh_leaf_${kind} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
 target_link_libraries(map_refresh_leaf_${kind} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
 target_link_options(map_refresh_leaf_${kind} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
endforeach()
target_compile_definitions(map_refresh_leaf_trace PRIVATE MAP_REFRESH_TRACE)
enable_testing()
add_test(NAME map_refresh_leaf_trace_launch COMMAND map_refresh_leaf_trace)
add_test(NAME map_refresh_leaf_plain_launch COMMAND map_refresh_leaf_plain)
