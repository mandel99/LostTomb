cmake_minimum_required(VERSION 3.20)
project(actor_coordinates_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(actor_coordinates_verify "${RESTORE}/tests/actor_coordinates.c" "${RESTORE}/tests/actor_coordinates_real_offset.c" "${RESTORE}/tests/actor_coordinates_real_inverse.c" "${RESTORE}/tests/actor_coordinates_real_world.c" "${RESTORE}/tests/actor_coordinates_real_actor.c")
target_include_directories(actor_coordinates_verify PRIVATE "${RESTORE}/include")
target_compile_options(actor_coordinates_verify PRIVATE /W4 "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
target_compile_definitions(actor_coordinates_verify PRIVATE _CRT_SECURE_NO_WARNINGS ACTOR_COORDINATES_STAGED)
set_property(TARGET actor_coordinates_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(actor_coordinates_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(actor_coordinates_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME actor_coordinates_smoke COMMAND actor_coordinates_verify)
