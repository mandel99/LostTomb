cmake_minimum_required(VERSION 3.20)
project(tool_selector_input_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(tool_selector_input_verify "${RESTORE}/tests/tool_selector_input.c" "${RESTORE}/tests/tool_selector_input_real.c")
target_include_directories(tool_selector_input_verify PRIVATE "${RESTORE}/include")
target_compile_options(tool_selector_input_verify PRIVATE /W4 "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
target_compile_definitions(tool_selector_input_verify PRIVATE _CRT_SECURE_NO_WARNINGS TOOL_SELECTOR_INPUT_STAGED)
set_property(TARGET tool_selector_input_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(tool_selector_input_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(tool_selector_input_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME tool_selector_input_smoke COMMAND tool_selector_input_verify)
