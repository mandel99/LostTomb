cmake_minimum_required(VERSION 3.20)
project(actor_predicate_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
  add_executable(actor_predicate_${kind} "${RESTORE}/tests/actor_predicate.c" "${RESTORE}/src/unit_00560.c" "${RESTORE}/tests/actor_predicate_helpers.c")
  target_include_directories(actor_predicate_${kind} PRIVATE "${RESTORE}/include")
  target_compile_options(actor_predicate_${kind} PRIVATE /O2 /W4)
  target_compile_definitions(actor_predicate_${kind} PRIVATE _CRT_SECURE_NO_WARNINGS)
  set_property(TARGET actor_predicate_${kind} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(actor_predicate_${kind} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
  target_link_options(actor_predicate_${kind} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
endforeach()
target_compile_definitions(actor_predicate_trace PRIVATE RESTORE_ACTOR_PREDICATE_TRACE_READS)
enable_testing()
add_test(NAME actor_predicate_trace_smoke COMMAND actor_predicate_trace)
add_test(NAME actor_predicate_plain_smoke COMMAND actor_predicate_plain)
