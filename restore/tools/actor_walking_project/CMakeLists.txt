cmake_minimum_required(VERSION 3.20)
project(actor_walking_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(actor_walking_verify "${RESTORE}/tests/actor_walking.c" "${RESTORE}/tests/actor_walking_real.c")
target_include_directories(actor_walking_verify PRIVATE "${RESTORE}/artifacts/actor_combat_phase/include" "${RESTORE}/include")
target_compile_options(actor_walking_verify PRIVATE /W4 "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
target_compile_definitions(actor_walking_verify PRIVATE _CRT_SECURE_NO_WARNINGS ACTOR_WALKING_STAGED)
set_property(TARGET actor_walking_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(actor_walking_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(actor_walking_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME actor_walking_smoke COMMAND actor_walking_verify)
