cmake_minimum_required(VERSION 3.20)
project(mirrored_raster_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(mirrored_raster_verify "${RESTORE}/tests/mirrored_raster.c" "${RESTORE}/tests/mirrored_raster_real_skip.c" "${RESTORE}/tests/mirrored_raster_plain.c" "${RESTORE}/artifacts/mirrored_raster_phase/unit_00131.c")
target_include_directories(mirrored_raster_verify PRIVATE "${RESTORE}/include")
target_compile_options(mirrored_raster_verify PRIVATE /O2 /W4)
target_compile_definitions(mirrored_raster_verify PRIVATE _CRT_SECURE_NO_WARNINGS RESTORE_MIRRORED_RASTER_TRACE MIRRORED_RASTER_STAGED)
set_property(TARGET mirrored_raster_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(mirrored_raster_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(mirrored_raster_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME mirrored_raster_smoke COMMAND mirrored_raster_verify)
