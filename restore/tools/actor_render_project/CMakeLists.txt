cmake_minimum_required(VERSION 3.20)
project(actor_render_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(actor_render_verify "${RESTORE}/tests/actor_render.c" "${RESTORE}/tests/actor_render_real_renderer.c" "${RESTORE}/tests/actor_render_real_graphics.c" "${RESTORE}/tests/actor_render_real_clips.c" "${RESTORE}/artifacts/actor_render_phase/unit_00562.c")
target_include_directories(actor_render_verify PRIVATE "${RESTORE}/include")
target_compile_options(actor_render_verify PRIVATE /O2 /W4)
target_compile_definitions(actor_render_verify PRIVATE _CRT_SECURE_NO_WARNINGS ACTOR_RENDER_STAGED)
set_property(TARGET actor_render_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(actor_render_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(actor_render_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME actor_render_smoke COMMAND actor_render_verify)
