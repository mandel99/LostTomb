cmake_minimum_required(VERSION 3.20)
project(ground_render_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(ground_render_verify "${RESTORE}/tests/ground_render.c" "${RESTORE}/tests/ground_render_real_graphics.c" "${RESTORE}/tests/ground_render_real_upper.c" "${RESTORE}/tests/ground_render_real_clips.c" "${RESTORE}/tests/ground_render_real_helpers.c" "${RESTORE}/artifacts/ground_render_phase/unit_00469.c" "${RESTORE}/artifacts/ground_render_phase/unit_00470.c" "${RESTORE}/artifacts/ground_render_phase/unit_00472.c")
target_include_directories(ground_render_verify PRIVATE "${RESTORE}/include")
target_compile_options(ground_render_verify PRIVATE /O2 /W4)
target_compile_definitions(ground_render_verify PRIVATE _CRT_SECURE_NO_WARNINGS GROUND_RENDER_VERIFY GROUND_RENDER_STAGED)
set_property(TARGET ground_render_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(ground_render_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(ground_render_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME ground_render_smoke COMMAND ground_render_verify)
