cmake_minimum_required(VERSION 3.20)
project(lookup_sentinel_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(lookup_sentinel_verify "${RESTORE}/tests/lookup_sentinel.c" "${RESTORE}/tests/lookup_sentinel_real_clear.c" "${RESTORE}/artifacts/lookup_sentinel_phase/unit_00459.c")
target_include_directories(lookup_sentinel_verify PRIVATE "${RESTORE}/include")
target_compile_options(lookup_sentinel_verify PRIVATE /O2 /W4)
target_compile_definitions(lookup_sentinel_verify PRIVATE _CRT_SECURE_NO_WARNINGS LOOKUP_SENTINEL_STAGED)
set_property(TARGET lookup_sentinel_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(lookup_sentinel_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(lookup_sentinel_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME lookup_sentinel_smoke COMMAND lookup_sentinel_verify)
