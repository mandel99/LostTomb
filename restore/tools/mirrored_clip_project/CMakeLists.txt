cmake_minimum_required(VERSION 3.20)
project(mirrored_clip_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(mirrored_clip_verify "${RESTORE}/tests/mirrored_clip.c" "${RESTORE}/tests/mirrored_clip_real_skip.c" "${RESTORE}/tests/mirrored_clip_plain132.c" "${RESTORE}/tests/mirrored_clip_plain133.c" "${RESTORE}/artifacts/mirrored_clip_phase/unit_00132.c" "${RESTORE}/artifacts/mirrored_clip_phase/unit_00133.c")
target_include_directories(mirrored_clip_verify PRIVATE "${RESTORE}/include")
target_compile_options(mirrored_clip_verify PRIVATE /O2 /W4)
target_compile_definitions(mirrored_clip_verify PRIVATE _CRT_SECURE_NO_WARNINGS RESTORE_MIRRORED_CLIP_TRACE MIRRORED_CLIP_STAGED)
set_property(TARGET mirrored_clip_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(mirrored_clip_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(mirrored_clip_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME mirrored_clip_smoke COMMAND mirrored_clip_verify)
