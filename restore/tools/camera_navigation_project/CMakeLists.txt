cmake_minimum_required(VERSION 3.20)
project(camera_navigation_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(camera_navigation_verify "${RESTORE}/tests/camera_navigation.c" "${RESTORE}/tests/camera_navigation_real.c")
target_include_directories(camera_navigation_verify PRIVATE "${RESTORE}/include")
target_compile_options(camera_navigation_verify PRIVATE /W4 "$<$<CONFIG:Release>:/O2>" "$<$<CONFIG:Debug>:/Od>")
target_compile_definitions(camera_navigation_verify PRIVATE _CRT_SECURE_NO_WARNINGS CAMERA_NAVIGATION_STAGED)
set_property(TARGET camera_navigation_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(camera_navigation_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(camera_navigation_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME camera_navigation_smoke COMMAND camera_navigation_verify)
