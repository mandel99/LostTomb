cmake_minimum_required(VERSION 3.20)
project(world_render_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(WORLD_RENDER_SOURCE "${RESTORE}/artifacts/world_render_phase/unit_00547.c" CACHE FILEPATH "Staged or applied world renderer")
add_executable(world_render_verify "${RESTORE}/tests/world_render.c" "${RESTORE}/tests/world_render_real_upper.c" "${RESTORE}/tests/world_render_real_clips.c" "${RESTORE}/tests/world_render_real_members.c" "${WORLD_RENDER_SOURCE}")
target_include_directories(world_render_verify PRIVATE "${RESTORE}/include")
target_compile_options(world_render_verify PRIVATE /O2 /W4)
target_compile_definitions(world_render_verify PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_definitions(world_render_verify PRIVATE WORLD_RENDER_STAGED)
set_property(TARGET world_render_verify PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
target_link_libraries(world_render_verify PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
target_link_options(world_render_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
enable_testing()
add_test(NAME world_render_smoke COMMAND world_render_verify)
