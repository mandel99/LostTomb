cmake_minimum_required(VERSION 3.20)
project(viewport_controller_verification C)
get_filename_component(RESTORE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(kind IN ITEMS trace plain)
  add_executable(viewport_controller_${kind} "${RESTORE}/tests/viewport_controller.c" "${RESTORE}/tests/viewport_controller_empty_callback.c" "${RESTORE}/src/unit_00468.c" "${RESTORE}/src/unit_01878.c")
  target_include_directories(viewport_controller_${kind} PRIVATE "${RESTORE}/include")
  target_compile_options(viewport_controller_${kind} PRIVATE /O2 /W4)
  target_compile_definitions(viewport_controller_${kind} PRIVATE _CRT_SECURE_NO_WARNINGS)
  set_property(TARGET viewport_controller_${kind} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
  target_link_libraries(viewport_controller_${kind} PRIVATE "${RESTORE}/build/c-only-debug/Debug/restore_data.lib" advapi32)
  target_link_options(viewport_controller_${kind} PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED)
endforeach()
target_compile_definitions(viewport_controller_trace PRIVATE RESTORE_VIEWPORT_CONTROLLER_TRACE)
enable_testing()
add_test(NAME viewport_controller_trace_smoke COMMAND viewport_controller_trace)
add_test(NAME viewport_controller_plain_smoke COMMAND viewport_controller_plain)
