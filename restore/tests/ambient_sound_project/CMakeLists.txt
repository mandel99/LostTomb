cmake_minimum_required(VERSION 3.25)
project(AmbientSound LANGUAGES C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(RESTORE_ROOT "../.." ABSOLUTE)
add_library(ambient_rng OBJECT "${RESTORE_ROOT}/src/unit_02825.c")
target_compile_definitions(ambient_rng PRIVATE FUN_00562348=ambient_rng_impl)
target_include_directories(ambient_rng PRIVATE "${RESTORE_ROOT}/include")
add_library(ambient_coord OBJECT "${RESTORE_ROOT}/src/unit_00461.c")
target_compile_definitions(ambient_coord PRIVATE FUN_00430960=ambient_coord_impl)
target_include_directories(ambient_coord PRIVATE "${RESTORE_ROOT}/include")
add_executable(restore_ambient_sound_verify "${RESTORE_ROOT}/tests/ambient_sound.c" "${RESTORE_ROOT}/src/unit_02486.c" $<TARGET_OBJECTS:ambient_rng> $<TARGET_OBJECTS:ambient_coord>)
target_include_directories(restore_ambient_sound_verify PRIVATE "${RESTORE_ROOT}/include")
target_compile_definitions(restore_ambient_sound_verify PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(restore_ambient_sound_verify PRIVATE /W4 $<$<CONFIG:Debug>:/Od>)
target_link_libraries(restore_ambient_sound_verify PRIVATE "${RESTORE_ROOT}/build/c-only-debug/Debug/restore_data.lib")
target_link_options(restore_ambient_sound_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
