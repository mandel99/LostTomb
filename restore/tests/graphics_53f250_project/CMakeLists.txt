cmake_minimum_required(VERSION 3.25)
project(Graphics53f250 LANGUAGES C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(RESTORE_ROOT "../.." ABSOLUTE)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use MSVC Win32.")
endif()
add_library(graphics_53f250_clippers OBJECT "${RESTORE_ROOT}/src/unit_01725.c" "${RESTORE_ROOT}/src/unit_01726.c")
target_include_directories(graphics_53f250_clippers PRIVATE "${RESTORE_ROOT}/include")
target_compile_definitions(graphics_53f250_clippers PRIVATE FUN_004cdad0=graphics_53f250_hclip_impl FUN_004cdb80=graphics_53f250_vclip_impl)

target_include_directories(graphics_53f250_verify PRIVATE "${RESTORE_ROOT}/include")
target_compile_definitions(graphics_53f250_verify PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(graphics_53f250_verify PRIVATE /W4 /Od)
target_link_libraries(graphics_53f250_verify PRIVATE "${RESTORE_ROOT}/build/c-only-debug/Debug/restore_data.lib")
target_link_options(graphics_53f250_verify PRIVATE /BASE:0x10000 /DYNAMICBASE:NO /FIXED /MAP)
